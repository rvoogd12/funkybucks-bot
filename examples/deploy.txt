name: Deploy to Orihost

on:
  push:
    branches: [main]

jobs:
  deploy:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repo
        uses: actions/checkout@v4

      - name: Deploy via SFTP (rsync over SSH)
        env:
          SFTP_HOST: ${{ secrets.ORIHOST_SFTP_HOST }}
          SFTP_USER: ${{ secrets.ORIHOST_SFTP_USER }}
          SFTP_PASSWORD: ${{ secrets.ORIHOST_SFTP_PASSWORD }}
          SFTP_PORT: ${{ secrets.ORIHOST_SFTP_PORT }}
          REMOTE_DIR: ${{ secrets.ORIHOST_REMOTE_DIR }}
        run: |
          set -e
          HOST="${SFTP_HOST#sftp://}"
          HOST="${HOST%/}"
          echo "Deploying to $HOST:$SFTP_PORT -> $REMOTE_DIR"
          sudo apt-get update
          sudo apt-get install -y lftp
          echo "Starting lftp mirror to $HOST..."
          lftp -u "$SFTP_USER,$SFTP_PASSWORD" -p ${SFTP_PORT:-22} "sftp://$HOST" \
            -e "set sftp:auto-confirm yes; set net:timeout 30; mirror --reverse --delete ./ \"$REMOTE_DIR\"; bye"
